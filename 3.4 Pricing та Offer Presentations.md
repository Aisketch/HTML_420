function announceToScreenReader(message) {
  const announcement = document.createElement('div');
  announcement.setAttribute('role', 'status');
  announcement.setAttribute('aria-live', 'polite');
  announcement.className = 'sr-only';
  announcement.textContent = message;

  setTimeout(() => announcement.remove(), 1000);
}

## 💻 CODE SNIPPETS & TEMPLATES

### Real-Time Price Calculator з Currency.js

```javascript
import currency from 'currency.js';

class PricingCalculator {
  constructor() {
    this.basePrice = 29;
    this.userPrice = 5;
    this.addons = { analytics: 15, api: 20, support: 10 };
    this.init();
  }
  
  init() {
    document.getElementById('userCount')?.addEventListener('input', 
      this.debounce(() => this.calculate(), 300)
    );
    
    document.querySelectorAll('.addon-checkbox').forEach(checkbox => {
      checkbox.addEventListener('change', () => this.calculate());
    });
    
    this.calculate();
  }
  
  calculate() {
    const users = parseInt(document.getElementById('userCount')?.value) || 1;
    const selectedAddons = Array.from(
      document.querySelectorAll('.addon-checkbox:checked')
    ).map(cb => cb.value);
    
    let total = currency(this.basePrice);
    
    if (users > 1) {
      total = total.add(currency(users - 1).multiply(this.userPrice));
    }
    
    selectedAddons.forEach(addon => {
      if (this.addons[addon]) {
        total = total.add(this.addons[addon]);
      }
    });
    
    this.updateDisplay(total);
  }
  
  updateDisplay(total) {
    const display = document.getElementById('totalPrice');
    if (!display) return;
    
    display.classList.add('updating');
    setTimeout(() => {
      display.textContent = total.format();
      display.classList.remove('updating');
      this.announcePrice(total.value);
    }, 150);
  }
  
  announcePrice(value) {
    const announcement = document.getElementById('price-announcement');
    if (announcement) {
      announcement.textContent = `Total: ${currency(value).format()} per month`;
    }
  }
  
  debounce(func, wait) {
    let timeout;
    return function(...args) {
      clearTimeout(timeout);
      timeout = setTimeout(() => func.apply(this, args), wait);
    };
  }
}

document.addEventListener('DOMContentLoaded', () => new PricingCalculator());
```

## ⚠️ ТИПОВІ ПОМИЛКИ ТА ЯК ЇХ УНИКНУТИ

### 1. Touch Target Size Violations

**❌ Помилка:** Buttons < 44×44px (iOS) або 48×48px (Android)

**✅ Рішення:**
```css
.pricing-button {
  min-width: 44px;
  min-height: 44px;
  padding: 12px 24px;
}

/* Extend hit area */
.small-icon::after {
  content: '';
  position: absolute;
  inset: -8px;
}
```

### 2. Poor Table Semantics

**❌ Помилка:** `<div>` з CSS display: table

**✅ Рішення:**
```html
<table>
  <thead>
    <tr>
      <th scope="col">Feature</th>
      <th scope="col">Basic</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <th scope="row">Storage</th>
      <td>10 GB</td>
    </tr>
  </tbody>
</table>
```

### 3. Missing Screen Reader Context

**❌ Помилка:** Strikethrough prices без context

**✅ Рішення:**
```html
<div class="price">
  <del aria-hidden="true">$99.00</del>
  <span class="sr-only">Original price: $99.00.</span>
  <ins aria-hidden="true">$79.00</ins>
  <span class="sr-only">Sale price: $79.00. Save 20%.</span>
</div>
```

### 4. Color Contrast Failures

**❌ Помилка:** #999 на white = 2.85:1 (fails)

**✅ Рішення:**
```css
.price-label {
  color: #595959; /* 7.0:1 - passes AAA */
  background: #FFFFFF;
}
```

### 5. JavaScript Float Errors

**❌ Помилка:** 0.1 + 0.2 = 0.30000000000000004

**✅ Рішення:**
```javascript
import currency from 'currency.js';
currency(0.1).add(0.2).format(); // "$0.30"
```

**Focus Management:**
```javascript
class AccessibleModal {
  open() {
    this.previousFocus = document.activeElement;
    this.modal.focus();
    this.trapFocus();
  }
  
  close() {
    this.previousFocus?.focus();
  }
  
  trapFocus() {
    this.modal.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') this.close();
      if (e.key === 'Tab') this.handleTab(e);
    });
  }
}
```
