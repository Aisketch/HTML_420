### Mobile-first implementation requires specific technical patterns

Performance optimization through lazy loading reduces page size by 60% and improves PageSpeed scores by 10-13 points on mobile. The critical technique combines loading="lazy" iframe attributes with Intersection Observer API:

```html
<iframe src="https://www.google.com/maps/embed?pb=..." 
        loading="lazy" width="600" height="450">
</iframe>
```

More sophisticated implementations use Intersection Observer to load maps only when visible:

```javascript
const mapObserver = new IntersectionObserver((entries, observer) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const iframe = document.createElement('iframe');
      iframe.src = entry.target.dataset.src;
      iframe.width = '100%';
      iframe.height = '100%';
      entry.target.appendChild(iframe);
      observer.unobserve(entry.target);
    }
  });
}, { rootMargin: '50px' });

document.querySelectorAll('.lazy-map').forEach(el => mapObserver.observe(el));
```

**Responsive design patterns prevent common mobile failures**. Maps must use fluid width containers with aspect ratio padding to maintain proportions across devices:


### Security mistakes cost hundreds of dollars daily

**Unrestricted API keys result in fraudulent charges within days**. Google Maps requires five security layers: HTTP referrer restrictions, API restrictions limiting which APIs the key can access, digital signatures for Static Maps/Street View, quota limits capping daily requests, and billing alerts monitoring for usage spikes. One compromised key can generate thousands in charges before detection.

**Best practices mandate separate keys per platform** (web, mobile, server) stored in environment variables, never committed to GitHub. Implement backend proxies to hide keys from client-side JavaScript:

```javascript
// Backend (Node.js/Express)
app.get('/api/maps', async (req, res) => {
  const apiKey = process.env.MAPS_API_KEY;
  const { location } = req.query;
  const response = await fetch(
    `https://maps.googleapis.com/maps/api/geocode/json?address=${location}&key=${apiKey}`
  );
  res.json(await response.json());
});

// Frontend - key never exposed
const location = await fetch('/api/maps?location=London');
```

Mapbox security requires URL-restricted tokens, scoped tokens limiting permissions, and immediate rotation if exposed. Both platforms support domain restrictions preventing unauthorized use.

### Performance optimization prevents rate limit disasters

**Caching strategies reduce API calls by 90%+**. Recommended cache durations: current weather 10-30 minutes, forecasts 1-6 hours, historical data 24 hours. LocalStorage implementation:

```javascript
function getCachedWeather(location, maxAge = 600000) {
  const cached = localStorage.getItem(`weather-${location}`);
  const timestamp = localStorage.getItem(`weather-${location}-time`);
  
  if (cached && Date.now() - timestamp < maxAge) {
    return JSON.parse(cached);
  }
  return null;
}
```

**Debouncing prevents excessive calls** from user input:

```javascript
function debounce(func, delay) {
  let timeoutId;
  return function(...args) {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => func.apply(this, args), delay);
  };
}

const debouncedSearch = debounce((value) => fetchWeather(value), 500);
searchInput.addEventListener('input', (e) => debouncedSearch(e.target.value));
```

**Exponential backoff handles rate limits gracefully**:

```javascript
async function fetchWithRetry(url, retries = 3) {
  let delay = 100;
  for (let i = 0; i < retries; i++) {
    try {
      const response = await fetch(url);
      if (response.status === 429) {
        await new Promise(resolve => setTimeout(resolve, delay));
        delay *= 2;
        continue;
      }
      return response;
    } catch (error) {
      if (i === retries - 1) throw error;
    }
  }
}
```

### React and Vue integration patterns for SPAs

**React hook pattern with caching**:

```javascript
import { useState, useEffect } from 'react';

function useWeather(location, cacheTime = 600000) {
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    const fetchWeather = async () => {
      try {
        const cached = localStorage.getItem(`weather-${location}`);
        const timestamp = localStorage.getItem(`weather-${location}-time`);
        
        if (cached && Date.now() - timestamp < cacheTime) {
          setWeather(JSON.parse(cached));
          setLoading(false);
          return;
        }
        
        const response = await fetch(`/api/weather?location=${location}`);
        if (!response.ok) throw new Error('Weather fetch failed');
        
        const data = await response.json();
        setWeather(data);
        localStorage.setItem(`weather-${location}`, JSON.stringify(data));
        localStorage.setItem(`weather-${location}-time`, Date.now());
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };
    
    fetchWeather();
  }, [location, cacheTime]);
  
  return { weather, loading, error };
}
```

**Vue 3 composable pattern**:

```javascript
import { ref, onMounted } from 'vue';

export function useWeatherWidget(location) {
  const weatherData = ref(null);
  const isLoading = ref(true);
  const error = ref(null);
  
  const fetchWeather = async () => {
    try {
      const response = await fetch(`/api/weather?q=${location}`);
      if (!response.ok) throw new Error('Fetch failed');
      weatherData.value = await response.json();
    } catch (err) {
      error.value = err.message;
    } finally {
      isLoading.value = false;
    }
  };
  
  onMounted(fetchWeather);
  
  return { weatherData, isLoading, error, fetchWeather };
}
```

### JavaScript implementation examples

**Vanilla JavaScript basic implementation**:

```javascript
async function loadReviews(productId) {
  const response = await fetch(`/api/reviews/${productId}`);
  const reviews = await response.json();
  
  const container = document.getElementById('reviews');
  reviews.forEach(review => {
    const div = document.createElement('div');
    div.className = 'review-card';
    div.innerHTML = `
      <div class="rating" aria-label="${review.rating} out of 5 stars">
        ${'★'.repeat(review.rating)}${'☆'.repeat(5-review.rating)}
      </div>
      <p class="review-text">${review.text}</p>
      <span class="reviewer">${review.name}</span>
    `;
    container.appendChild(div);
  });
}
```

**React component with lazy loading**:

```javascript
import { useState, useEffect } from 'react';

function ReviewWidget({ productId }) {
  const [reviews, setReviews] = useState([]);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    const observer = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting) {
        fetch(`/api/reviews/${productId}`)
          .then(res => res.json())
          .then(data => {
            setReviews(data);
            setLoading(false);
          });
        observer.disconnect();
      }
    }, { rootMargin: '100px' });
    
    observer.observe(document.getElementById('review-trigger'));
    
    return () => observer.disconnect();
  }, [productId]);
  
  if (loading) return <div id="review-trigger">Loading reviews...</div>;
  
  return (
    <div className="reviews">
      {reviews.map(review => (
        <div key={review.id} className="review-card">
          <div className="rating" aria-label={`${review.rating} out of 5 stars`}>
            {'★'.repeat(review.rating)}{'☆'.repeat(5-review.rating)}
          </div>
          <p>{review.text}</p>
          <span>{review.name}</span>
        </div>
      ))}
    </div>
  );
}
```
