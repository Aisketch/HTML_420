class Router {
  constructor(routes) {
    this.routes = routes;
    window.addEventListener('popstate', () => this.route());
    document.addEventListener('click', e => {
      if (e.target.matches('[data-link]')) {
        e.preventDefault();
        this.navigate(e.target.href);
      }

    history.pushState({}, '', path);
    await this.route();

**1. Progressive Validation**
```javascript
class FormValidator {
  constructor(form) {
    this.form = form;
    this.hasSubmitted = false;
    this.errors = new Map();
    
    form.addEventListener('submit', e => {
      e.preventDefault();
      this.hasSubmitted = true;
      if (this.validateAll()) form.submit();
    });
    
    form.querySelectorAll('input').forEach(input => {
      input.addEventListener('blur', () => {
        if (this.hasSubmitted) this.validateField(input);
      });
      
      input.addEventListener('input', () => {
        if (this.errors.has(input.name)) this.clearError(input);
      });
    });
  }
  
  validateField(input) {
    if (!input.checkValidity()) {
      this.setError(input, input.validationMessage);
      return false;
    }
    this.clearError(input);
    return true;
  }
  
  setError(input, msg) {
    this.errors.set(input.name, msg);
    input.setAttribute('aria-invalid', 'true');
    const error = document.getElementById(`${input.id}-error`);
    error.textContent = msg;
    error.hidden = false;
  }
  
  clearError(input) {
    this.errors.delete(input.name);
    input.setAttribute('aria-invalid', 'false');
  }
}
```

```javascript
let step = 1;
const totalSteps = 3;

document.getElementById('next').addEventListener('click', async () => {
  const fieldset = document.querySelector(`[data-step="${step}"]`);
  const inputs = fieldset.querySelectorAll('input[required]');
  
  let valid = true;
  inputs.forEach(input => {
    if (!input.checkValidity()) {
      valid = false;
      input.reportValidity();
    }
  });
  
  if (valid && step < totalSteps) {
    fieldset.hidden = true;
    step++;
    document.querySelector(`[data-step="${step}"]`).hidden = false;
    updateProgress();
  }
});

  document.getElementById('progress').textContent = 
    `Step ${step} of ${totalSteps}`;
}
```

1. **Native Support:** loading="lazy" 95% browser support (Safari 16.4+)
2. **⚠️ LCP Critical:** НІКОЛИ не lazy load LCP images - погіршує на 600-1200ms
3. **Performance Gain:** 4-30% LCP improvement з правильною імплементацією
4. **Bundle Sizes:** lozad 1.25KB, vanilla-lazyload 1.9KB, lazysizes 10KB
5. **Mobile Testing:** 97.5% images load within 10ms visibility на 3G
6. **fetchpriority:** Firefox support з жовтня 2024, full coverage
7. **Intersection Observer:** Best practice для custom lazy loading
8. **CLS Prevention:** Завжди встановлюйте width/height
9. **Progressive Loading:** LQIP, progressive JPEG, WebP/AVIF
10. **Network Aware:** Adjust rootMargin: 4G=1250px, 3G=2500px

### ⭐ BEST PRACTICES

**1. Native Lazy Loading (Best)**
```html
<!-- Hero/LCP - EAGER load -->
<img
  src="hero.jpg"
  fetchpriority="high"
  alt="Hero"
  width="1920"
  height="1080"
/>

<!-- Above fold - normal -->
<img
  src="logo.jpg"
  alt="Logo"
  width="200"
  height="100"
/>

<!-- Below fold - LAZY -->
<img
  src="product.jpg"
  loading="lazy"
  decoding="async"
  alt="Product"
  width="800"
  height="600"
/>
```

**2. Intersection Observer Custom**
```javascript
const imageObserver = new IntersectionObserver((entries, observer) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src;
      img.classList.remove('lazy');
      observer.unobserve(img); // Важливо!
    }
  });
}, {
  rootMargin: '50px', // Preload before visible
  threshold: 0
});

document.querySelectorAll('img.lazy').forEach(img => {
  imageObserver.observe(img);
});
```

**3. Progressive Blur-Up**
```html
<img
  src="tiny-placeholder.jpg"
  data-src="full-image.jpg"
  class="progressive blur"
  width="1600"
  height="900"
/>

<style>
.progressive {
  transition: filter 0.3s;
}
.progressive.blur {
  filter: blur(10px);
  transform: scale(1.05);
}
</style>

<script>
const observer = new IntersectionObserver(entries => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      const full = new Image();
      full.onload = () => {
        img.src = full.src;
        img.classList.remove('blur');
      };
      full.src = img.dataset.src;
      observer.unobserve(img);
    }
  });
});
</script>
```

```html

>

<script>
const videoObserver = new IntersectionObserver(entries => {

```

## ВИСНОВОК

Це комплексне дослідження JavaScript рішень для Mobile First Single Page development демонструє, що у 2024-2025 розробники мають потужний арсенал інструментів і патернів. **Vanilla JS** залишається життєздатним для багатьох проектів завдяки сучасним ES features. **Мікро-бібліотеки** як Alpine.js та htmx забезпечують оптимальний баланс функціональності та розміру. **GSAP** продовжує домінувати в складних анімаціях. **Form validation** з правильними бібліотеками та патернами значно покращує UX. **Lazy loading** критично важливий для mobile performance, але потребує обережності з LCP images.

Ключ до успіху - **Mobile First thinking** на кожному етапі, дотримання **Performance Budget**, забезпечення **WCAG 2.1 AA accessibility**, та continuous testing на реальних пристроях. Правильне поєднання цих п'яти підетапів створює foundation для висококонверсійних, швидких та доступних Single Page додатків.

**Найважливіше:** Починайте з найпростішого рішення (Vanilla JS, native lazy loading, HTML5 validation), і додавайте складність тільки коли це необхідно. Progressive enhancement та accessibility - не опції, а обов'язкові вимоги для професійної розробки у 2025 році.
